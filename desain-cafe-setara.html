Database (MySQL)

Struktur tabel dan relasi. PK = primary key, FK = foreign key.

users

  • idBIGINT
  • namaVARCHAR(100)
  • emailVARCHAR(150) UQ
  • passwordVARCHAR(255)
  • google_idVARCHAR(100) NULL
  • roleENUM(owner,accounting,kasir)

kategori

  • idBIGINT
  • namaVARCHAR(50)

menu

  • idBIGINT
  • kategori_idFK → kategori
  • namaVARCHAR(100)
  • harga_jualDECIMAL(12,0)
  • fotoVARCHAR(255) NULL
  • aktifBOOLEAN

bahan

  • idBIGINT
  • namaVARCHAR(100)
  • satuanVARCHAR(20)
  • harga_beli_per_satuanDECIMAL(12,2)
  • stokDECIMAL(12,2)
  • stok_minimumDECIMAL(12,2)

resep

  • idBIGINT
  • menu_idFK → menu
  • bahan_idFK → bahan
  • takaranDECIMAL(10,2)

transaksi

  • idBIGINT
  • kodeVARCHAR(20) UQ
  • user_idFK → users
  • tanggalDATETIME
  • totalDECIMAL(12,0)
  • metode_bayarENUM(tunai,qris,debit)

detail_transaksi

  • idBIGINT
  • transaksi_idFK → transaksi
  • menu_idFK → menu
  • qtyINT
  • harga_satuanDECIMAL(12,0)
  • hpp_satuanDECIMAL(12,0)

pergerakan_stok

  • idBIGINT
  • bahan_idFK → bahan
  • tipeENUM(masuk,keluar,penjualan,koreksi)
  • qtyDECIMAL(12,2)
  • transaksi_idFK NULL
  • tanggalDATETIME

kas

  • idBIGINT
  • user_idFK → users
  • tipeENUM(masuk,keluar)
  • kategoriVARCHAR(50)
  • jumlahDECIMAL(12,0)
  • keteranganTEXT NULL
  • tanggalDATETIME

Skema SQL (ringkas)

CREATE TABLE menu (
  id           BIGINT PRIMARY KEY AUTO_INCREMENT,
  kategori_id  BIGINT NOT NULL,
  nama         VARCHAR(100) NOT NULL,
  harga_jual   DECIMAL(12,0) NOT NULL,
  aktif        BOOLEAN DEFAULT TRUE,
  FOREIGN KEY (kategori_id) REFERENCES kategori(id)
);

CREATE TABLE resep (
  id        BIGINT PRIMARY KEY AUTO_INCREMENT,
  menu_id   BIGINT NOT NULL,
  bahan_id  BIGINT NOT NULL,
  takaran   DECIMAL(10,2) NOT NULL,
  FOREIGN KEY (menu_id)  REFERENCES menu(id),
  FOREIGN KEY (bahan_id) REFERENCES bahan(id)
);
